  jsCode={`const { createApp, ref, onMounted } = Vue;

const App = {
  setup() {
    const posts = ref([]);
    const loading = ref(false);
    const error = ref("");

    async function fetchPosts() {
      loading.value = true;
      error.value = "";

      try {
        const response = await fetch("https://jsonplaceholder.typicode.com/posts");

        if (!response.ok) {
          throw new Error(\`Failed to fetch posts: HTTP \${response.status}\`);
        }

        const data = await response.json();

        if (!Array.isArray(data)) {
          throw new Error("Invalid posts response");
        }

        posts.value = data;
      } catch (err) {
        error.value = err instanceof Error ? err.message : "Something went wrong";
      } finally {
        loading.value = false;
      }
    }

    onMounted(fetchPosts);

    return {
      posts,
      loading,
      error,
      fetchPosts,
    };
  },
};

createApp(App).mount("#app");`}

## 📌 Vue.js Key Concepts Revision Summary Table

| Topic | Key Concept |
| :--- | :--- |
| **Object.freeze()** | Prevents changes to top-level own properties; shallow operation |
| **Object Creation** | Object literals, constructor functions, ES classes, `Object.create()`, and `Object.create(null)` |
| **CSS Layout Behavior** | Layout differences between `display: none`, `visibility: hidden`, and `opacity: 0` |
| **Vue Lifecycle Hooks** | Mounting (`onMounted`), updates (`onUpdated`), and teardown cleanup (`onUnmounted`) |
| **Vue Components & Communication** | Props down (`defineProps`), events up (`defineEmits`), and one-way data flow |
| **Semantic HTML & Accessibility** | Meaningful document landmarks (`header`, `nav`, `main`, `article`), keyboard focus, and native elements |
| **JavaScript Event Loop** | Synchronous stack execution, microtask queue (Promises), and macrotask queue (`setTimeout`) |
| **Vue 3 Reactivity** | `ref()` for primitives/containers with `.value` vs `reactive()` Proxies and `toRefs()` destructuring |
| **Vue Directives** | `v-if`, `v-else-if`, `v-else`, `v-show`, `v-for` with stable `:key`, `v-model`, `v-bind`, and `v-on` |
| **CSS Libraries & Selection** | Utility-first (Tailwind CSS) vs Component Frameworks (Vuetify, Bootstrap) selection criteria |
| **Shallow vs Deep Copy** | Memory reference sharing in spread `{ ...obj }` vs native recursive cloning with `structuredClone()` |
| **Promises & Async/Await** | Pending/Fulfilled/Rejected states, `try/catch/finally` control flow, and error propagation |
| **Promise Combinators & Fetch** | `Promise.all()`, `allSettled()`, `race()`, `any()`, and checking `response.ok` on HTTP requests |
| **API Integration Challenge** | Vue 3 fetch state management, table rendering, retry triggers, and accessibility roles |
